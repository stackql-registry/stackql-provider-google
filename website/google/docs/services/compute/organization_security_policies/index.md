```sql
associations,
creationTimestamp,

fingerprint,
kind,
labelFingerprint,
labels,
parent,
recaptchaOptionsConfig,
region,
rules,
selfLink,
shortName,
type,
userDefinedFields
FROM google.compute.organization_security_policies
WHERE securityPolicy = '{{ securityPolicy }}' -- required
;
```
</TabItem>

```sql
SELECT
id,
items,
kind,
nextPageToken,
warning
FROM google.compute.organization_security_policies
```
