```sql
SELECT
name,
activeLearningConfig,
annotationLabels,
createTime,
currentSpend,
datasets,
displayName,
encryptionSpec,
error,
inputs,
inputsSchemaUri,
instructionUri,
labelerCount,
labelingProgress,
labels,
specialistPools,
state,
updateTime
FROM google.aiplatform.data_labeling_jobs
WHERE projectsId = '{{ projectsId }}' -- required
AND locationsId = '{{ locationsId }}' -- required
AND dataLabelingJobsId = '{{ dataLabelingJobsId }}' -- required
;
```
